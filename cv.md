## Code Examples

```
/**
 * Returns the sum of the digits of a given number.
 *
 * @param {number} num
 * @return {number}
 *
 * @example:
 *   123 => 6  // (1+2+3)
 *   202 => 4  // (2+0+2)
 *   5   => 5  // 5
 */
function getSumOfDigits(num) {
  const digitsArr = num.toString().split('').map(Number);
  const sum = digitsArr.reduce((acc, digit) => acc + digit, 0);
  return sum;
}
```
